// ------------------------------------------------------------------ web entry points
function doPost(e) {
  var req = {};
  try {
    var body = e && e.postData && e.postData.contents;
    if (body) req = JSON.parse(body);
  } catch (err) {
    req = {};
  }
  if (e && e.parameter) {
    Object.keys(e.parameter).forEach(function (k) { if (req[k] === undefined) req[k] = e.parameter[k]; });
  }
  return route_(req);
}

function doGet(e) {
  var p = (e && e.parameter) || {};
  if (!p.action) {
    return ContentService.createTextOutput('Money Tracker backend is running ✅').setMimeType(ContentService.MimeType.TEXT);
  }
  return route_(p);
}

function route_(req) {
  var store = new SheetStore_(ss_());
  var settings = MT.settingsMap(store.all('settings'));
  if (!settings.token || String(req.token || '') !== String(settings.token)) {
    return out_({ ok: false, error: 'unauthorized', message: '⛔ التوكن غلط' }, req.format);
  }
  var res;
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(25000);
    if (String(req.action || '').toLowerCase() === 'refreshrates') {
      var fr = fetchRates_(store);
      res = { ok: fr.ok, rate: fr.rate, source: fr.source, message: fr.ok ? 'USDT ' + fr.rate + ' (' + fr.source + ')' : 'تعذر تحديث السعر' };
    } else {
      res = MT.handleRequest(req, store, env_(store, settings));
    }
  } catch (err) {
    res = { ok: false, error: String(err && err.message || err), message: '⚠️ حصل خطأ: ' + String(err && err.message || err) };
  } finally {
    try { lock.releaseLock(); } catch (e2) { /* not held */ }
  }
  if (INGEST_ACTIONS.indexOf(String(req.action || '').toLowerCase()) >= 0) log_(req, res);
  return out_(res, req.format);
}

function out_(res, format) {
  if (format === 'text') {
    var text = res.message || res.text || (res.ok ? 'OK' : (res.error || 'error'));
    return ContentService.createTextOutput(text).setMimeType(ContentService.MimeType.TEXT);
  }
  return ContentService.createTextOutput(JSON.stringify(res)).setMimeType(ContentService.MimeType.JSON);
}

// ------------------------------------------------------------------ sheet store
function SheetStore_(ss) {
  this.ss = ss;
  this.cache = {};
}
SheetStore_.prototype.sheet = function (table) {
  var sh = this.ss.getSheetByName(SHEET_NAMES[table]);
  if (!sh) sh = ensureSheet_(this.ss, table);
  return sh;
};
SheetStore_.prototype.read = function (table) {
  if (this.cache[table]) return this.cache[table];
  var sh = this.sheet(table);
  var lastCol = Math.max(sh.getLastColumn(), 1);
  var header = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(String);
  var missing = MT.SCHEMA[table].filter(function (c) { return header.indexOf(c) < 0; });
  if (missing.length) {
    sh.getRange(1, header.length + 1, 1, missing.length).setValues([missing]);
    header = header.concat(missing);
  }
  var last = sh.getLastRow();
  var values = last > 1 ? sh.getRange(2, 1, last - 1, header.length).getValues() : [];
  var keyCol = table === 'settings' ? 'key' : 'id';
  var rows = [];
  var rowNums = [];
  for (var i = 0; i < values.length; i++) {
    var o = {};
    for (var j = 0; j < header.length; j++) o[header[j]] = fromCell_(header[j], values[i][j]);
    if (o[keyCol] === '' || o[keyCol] === null || o[keyCol] === undefined) continue;
    rows.push(o);
    rowNums.push(i + 2);
  }
  this.cache[table] = { header: header, rows: rows, rowNums: rowNums, keyCol: keyCol, sheet: sh };
  return this.cache[table];
};
SheetStore_.prototype.all = function (table) {
  var d = this.read(table);
  if (table === 'settings') {
    return d.rows.map(function (r) { return { key: String(r.key), value: r.value === null || r.value === undefined ? '' : r.value }; });
  }
  return d.rows.map(function (r) { return MT.coerce(table, r); });
};
SheetStore_.prototype.recent = function (table, n) {
  var all = this.all(table);
  return all.slice(Math.max(0, all.length - n));
};
SheetStore_.prototype.upsert = function (table, row) {
  var d = this.read(table);
  var key = row[d.keyCol];
  var idx = -1;
  for (var i = 0; i < d.rows.length; i++) if (String(d.rows[i][d.keyCol]) === String(key)) { idx = i; break; }
  var arr = d.header.map(function (h) { return toCell_(h, row[h] === undefined && idx >= 0 ? d.rows[idx][h] : row[h]); });
  if (idx >= 0) {
    d.sheet.getRange(d.rowNums[idx], 1, 1, arr.length).setValues([arr]);
    var merged = {};
    d.header.forEach(function (h) { merged[h] = row[h] === undefined ? d.rows[idx][h] : row[h]; });
    d.rows[idx] = merged;
  } else {
    d.sheet.appendRow(arr);
    var fresh = {};
    d.header.forEach(function (h) { fresh[h] = row[h]; });
    d.rows.push(fresh);
    d.rowNums.push(d.sheet.getLastRow());
  }
  return row;
};
SheetStore_.prototype.remove = function (table, id) {
  var d = this.read(table);
  for (var i = 0; i < d.rows.length; i++) {
    if (String(d.rows[i][d.keyCol]) === String(id)) {
      d.sheet.deleteRow(d.rowNums[i]);
      d.rows.splice(i, 1);
      d.rowNums.splice(i, 1);
      for (var k = i; k < d.rowNums.length; k++) d.rowNums[k] -= 1;
      return true;
    }
  }
  return false;
};

function fromCell_(field, v) {
  if (v instanceof Date) {
    if (MT.DATE_ONLY_FIELDS.indexOf(field) >= 0) return Utilities.formatDate(v, TZ, 'yyyy-MM-dd');
    return v.toISOString();
  }
  return v;
}
function toCell_(field, v) {
  if (v === null || v === undefined) return '';
  if (typeof v === 'number' || typeof v === 'boolean') return v;
  var s = String(v);
  // Never let text (e.g. an SMS) be read as a formula.
  if (/^[=+\-@]/.test(s) && !(field === 'value' && /^-?\d+(\.\d+)?$/.test(s))) s = "'" + s;
  return s;
}

function ensureSheet_(ss, table) {
  var name = SHEET_NAMES[table];
  var sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  var cols = MT.SCHEMA[table];
  if (sh.getLastRow() < 1) {
    sh.getRange(1, 1, 1, cols.length).setValues([cols]);
  }
  sh.getRange(1, 1, 1, Math.max(sh.getLastColumn(), cols.length)).setFontWeight('bold');
  sh.setFrozenRows(1);
  var header = sh.getRange(1, 1, 1, Math.max(sh.getLastColumn(), 1)).getValues()[0].map(String);
  header.forEach(function (h, i) {
    var isText = MT.NUM_FIELDS.indexOf(h) < 0 && MT.BOOL_FIELDS.indexOf(h) < 0 && !(table === 'settings' && h === 'value');
    if (isText) sh.getRange(1, i + 1, sh.getMaxRows(), 1).setNumberFormat('@');
  });
  return sh;
}

/** Upgrades older sheets: phone credit becomes its own currency, RedotPay gets its card fees, new category. */
function migrate_(store) {
  store.all('accounts').forEach(function (a) {
    var changed = false;
    if (a.kind === 'phone' && a.currency === 'EGP') { a.currency = 'CRD'; changed = true; }
    if (/redot/i.test(a.name + a.match) && !a.feePct && !a.fxFeePct) { a.feePct = 1; a.fxFeePct = 1.2; changed = true; }
    if (changed) store.upsert('accounts', MT.coerce('accounts', a));
  });
  var cats = store.all('categories');
  MT.defaultCategories().forEach(function (c) {
    if (!cats.some(function (x) { return x.id === c.id; })) store.upsert('categories', MT.coerce('categories', c));
  });
  var s = MT.settingsMap(store.all('settings'));
  var add = { creditNet: '0.70', creditSell: '1', usdPremium: '1', usdtRate: '', usdtSource: '', usdtAt: '' };
  Object.keys(add).forEach(function (k) { if (s[k] === undefined) store.upsert('settings', { key: k, value: add[k] }); });
}

/**
 * RedotPay is funded with USDT, so a dollar there costs the P2P (parallel-market) price, not the bank rate.
 * Tries Binance P2P, then OKX P2P. Stores usdtRate / usdtSource / usdtAt in Settings.
 */
function fetchRates_(store) {
  store = store || new SheetStore_(ss_());
  var rate = 0, source = '';
  var median = function (arr) {
    arr = arr.filter(function (x) { return x > 0; }).sort(function (a, b) { return a - b; });
    return arr.length ? arr[Math.floor(arr.length / 2)] : 0;
  };
  try {
    var r = UrlFetchApp.fetch('https://p2p.binance.com/bapi/c2c/v2/friendly/c2c/adv/search', {
      method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      payload: JSON.stringify({ asset: 'USDT', fiat: 'EGP', tradeType: 'BUY', page: 1, rows: 10, payTypes: [], publisherType: null })
    });
    var j = JSON.parse(r.getContentText());
    rate = median((j.data || []).map(function (d) { return Number(d.adv && d.adv.price); }));
    if (rate) source = 'Binance P2P';
  } catch (e) { /* try the next source */ }
  if (!rate) {
    try {
      var r2 = UrlFetchApp.fetch('https://www.okx.com/v3/c2c/tradingOrders/books?quoteCurrency=EGP&baseCurrency=USDT&side=sell&paymentMethod=all&userType=all', { muteHttpExceptions: true });
      var j2 = JSON.parse(r2.getContentText());
      rate = median(((j2.data || {}).sell || []).slice(0, 10).map(function (d) { return Number(d.price); }));
      if (rate) source = 'OKX P2P';
    } catch (e2) { /* fall back to bank rate + premium */ }
  }
  if (rate > 20 && rate < 500) {
    rate = Math.round(rate * 100) / 100;
    store.upsert('settings', { key: 'usdtRate', value: String(rate) });
    store.upsert('settings', { key: 'usdtSource', value: source });
    store.upsert('settings', { key: 'usdtAt', value: new Date().toISOString() });
    return { ok: true, rate: rate, source: source };
  }
  return { ok: false };
}

function daily() {
  var store = new SheetStore_(ss_());
  try { fetchRates_(store); } catch (e) { /* keep last rate */ }
  var settings = MT.settingsMap(store.all('settings'));
  var env = env_(store, settings);
  var lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try {
    MT.handleRequest({ action: 'tick' }, store, env);
    migrate_(store);
  } finally {
    lock.releaseLock();
  }
  if (String(settings.emailReminders || 'on') === 'off') return;
  var email = Session.getEffectiveUser().getEmail();
  if (!email) return;
  var r = MT.handleRequest({ action: 'reminders' }, store, env);
  if (r.count) MailApp.sendEmail(email, 'Money Tracker — تذكير', r.text);
  if (env.today.slice(8) === '01') {
    var lastMonth = MT.monthKey(MT.addDays(env.today, -1));
    MailApp.sendEmail(email, 'Money Tracker — ملخص ' + lastMonth, monthSummaryText_(store, lastMonth, env));
  }
}

function monthSummaryText_(store, month, env) {
  var txs = store.all('transactions');
  var cats = store.all('categories');
  var st = MT.monthStats(txs, month, env.rates, function (tx) { return Utilities.formatDate(new Date(tx.date), TZ, 'yyyy-MM-dd'); });
  var lines = ['ملخص شهر ' + month, '', 'دخل: ' + MT.fmtMoney(st.income, 'EGP'), 'صرفت: ' + MT.fmtMoney(st.spent, 'EGP'),
    'الفرق: ' + MT.fmtMoney(st.net, 'EGP'), 'رسوم بنكية: ' + MT.fmtMoney(st.fees, 'EGP'), '', 'أكتر حاجات صرفت عليها:'];
  Object.keys(st.byCategory).sort(function (a, b) { return st.byCategory[b] - st.byCategory[a]; }).slice(0, 6).forEach(function (id) {
    var c = cats.filter(function (x) { return x.id === id; })[0];
    lines.push('• ' + (c ? c.icon + ' ' + c.name : id) + ': ' + MT.fmtMoney(st.byCategory[id], 'EGP'));
  });
  lines.push('', 'الاشتراكات في الشهر: ' + MT.fmtMoney(MT.subsMonthlyTotal(store.all('subscriptions'), env.rates), 'EGP'));
  return lines.join('\n');
}

function log_(req, res) {
  try {
    var ss = ss_();
    var sh = ss.getSheetByName(LOG_SHEET);
    if (!sh) {
      sh = ss.insertSheet(LOG_SHEET);
      sh.getRange(1, 1, 1, 5).setValues([['time', 'action', 'input', 'result', 'message']]);
      sh.setFrozenRows(1);
      sh.getRange(1, 1, sh.getMaxRows(), 5).setNumberFormat('@');
    }
    var input = req.text || [req.amount, req.merchant, req.card, req.to, req.direction].filter(Boolean).join(' | ');
    var result = res.ignored ? 'ignored:' + (res.reason || '') : res.merged ? 'merged' : res.ok ? 'saved' : 'error';
    sh.appendRow([new Date().toISOString(), String(req.action), toCell_('input', String(input || '').slice(0, 500)), result, toCell_('m', res.message || res.error || '')]);
    var extra = sh.getLastRow() - 301;
    if (extra > 0) sh.deleteRows(2, extra);
  } catch (e) { /* logging must never break a request */ }
}
